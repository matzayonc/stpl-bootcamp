%% ---- Palette ----------------------------------------------------
%% Six roles. Keep the roles even if you change the hexes: the
%% preamble and every diagram recipe refer to them by role.
\definecolor{dpbg}{HTML}{0B0E0F}      % slide background
\definecolor{dpcard}{HTML}{161A1B}    % block / card surface
\definecolor{dpborder}{HTML}{1E2526}  % hairlines, progress-bar track
\definecolor{dpaccent}{HTML}{DC143C}  % brand accent: rules, bullets, keywords
\definecolor{dpfg}{HTML}{FFFFFF}      % primary text
\definecolor{dpmuted}{HTML}{8A9BA8}   % secondary text, axis lines, citations

%% Light-background decks: swap dpbg/dpfg, then darken dpmuted and LIGHTEN
%% dpcard and dpborder to match — all four were picked for a dark ground and
%% stay near-black otherwise (a black progress-bar track on white is the
%% giveaway). E.g.
%%   \definecolor{dpbg}{HTML}{FFFFFF}
%%   \definecolor{dpcard}{HTML}{F2F4F5}
%%   \definecolor{dpborder}{HTML}{E3E7E9}
%%   \definecolor{dpfg}{HTML}{111417}
%%   \definecolor{dpmuted}{HTML}{5A6B78}
